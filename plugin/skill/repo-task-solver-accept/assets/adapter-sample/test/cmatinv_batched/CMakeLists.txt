cmake_minimum_required(VERSION 3.16)
project(cmatinv_test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT OPS_SOLVER_REPO OR NOT DEFINED ENV{ASCEND_HOME_PATH})
  message(FATAL_ERROR "Set OPS_SOLVER_REPO and source CANN set_env.sh")
endif()
set(CANN "$ENV{ASCEND_HOME_PATH}")
add_library(cmatinv_adapter STATIC cmatinv_batched_adapter.cpp)
target_include_directories(cmatinv_adapter PUBLIC "${CMAKE_CURRENT_LIST_DIR}/../../include" "${OPS_SOLVER_REPO}/include" "${CANN}/${CMAKE_SYSTEM_PROCESSOR}-linux/include")
target_link_libraries(cmatinv_adapter PUBLIC "${OPS_SOLVER_REPO}/build/libops_solver.so" "${CANN}/lib64/libascendcl.so" ${CMAKE_DL_LIBS})
add_executable(cmatinv_test ../common/adapter_exec.cpp)
target_link_libraries(cmatinv_test PRIVATE cmatinv_adapter)
