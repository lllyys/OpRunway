cmake_minimum_required(VERSION 3.16)
project(cmatinv_adapter_example LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT OPS_SOLVER_REPO OR NOT SOLVER_ADAPTER_INCLUDE OR NOT DEFINED ENV{ASCEND_HOME_PATH})
  message(FATAL_ERROR "Set OPS_SOLVER_REPO, SOLVER_ADAPTER_INCLUDE and source CANN set_env.sh")
endif()
set(CANN "$ENV{ASCEND_HOME_PATH}")
add_library(cmatinv_adapter STATIC cmatinv_batched_adapter.cpp)
target_include_directories(cmatinv_adapter PUBLIC "${SOLVER_ADAPTER_INCLUDE}" "${OPS_SOLVER_REPO}/include" "${CANN}/${CMAKE_SYSTEM_PROCESSOR}-linux/include")
target_link_libraries(cmatinv_adapter PUBLIC "${OPS_SOLVER_REPO}/build/libops_solver.so" "${CANN}/lib64/libascendcl.so")
add_executable(cmatinv_adapter_test cmatinv_batched_test.cpp)
target_link_libraries(cmatinv_adapter_test PRIVATE cmatinv_adapter)
